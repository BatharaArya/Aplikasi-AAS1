(function(){

function load(i){if(typeof THREE!=='undefined'){start();return}if(i>=SRC.length){fail();return}var s=document.createElement('script');s.src=SRC[i];s.onload=function(){load(i+1)};s.onerror=function(){load(i+1)};document.head.appendChild(s)}

<script type="text/plain" id="app">

const $=s=>document.querySelector(s),T=THREE;



let S={wl:324.7, flame:true, aspirating:false, conc:2.0, beam:true, mode:0};

const Cu_Wavelength = 324.7;



const cv=$('#cv'),R=new T.WebGLRenderer({canvas:cv,antialias:true,alpha:true});

R.setClearColor(0x0b1220,1);R.shadowMap.enabled=true;

const scene=new T.Scene(),cam=new T.PerspectiveCamera(45,1,.1,100);

scene.add(new T.HemisphereLight(0xffffff,0x1a2233,.6));

const dl=new T.DirectionalLight(0xffffff,.8);dl.position.set(5,10,7);scene.add(dl);



/* --- AAS Models Components --- */

const root=new T.Group();scene.add(root);

const mats={

  body: new T.MeshStandardMaterial({color:0xe2e8f0, roughness:.3, metalness:.1}),

  dark: new T.MeshStandardMaterial({color:0x1e293b, roughness:.5}),

  burner: new T.MeshStandardMaterial({color:0x94a3b8, metalness:.8, roughness:.2}),

  flame: new T.MeshBasicMaterial({color:0x38bdf8, transparent:true, opacity:.75}),

  flameCu: new T.MeshBasicMaterial({color:0x34d399, transparent:true, opacity:.85}),

  hcl: new T.MeshStandardMaterial({color:0x38bdf8, transparent:true, opacity:.6, metalness:.3}),

  beam: new T.MeshBasicMaterial({color:0x38bdf8, transparent:true, opacity:.5}),

  atom: new T.MeshBasicMaterial({color:0xf97316})

};



// AAS Main Body

const body=new T.Mesh(new T.BoxGeometry(4.2, 2.2, 2.2), mats.body);

body.position.set(0, 1.1, 0); root.add(body);



// Front Compartment / Burner Chamber cutout area

const chamber=new T.Mesh(new T.BoxGeometry(1.4, 1.2, 1.2), mats.dark);

chamber.position.set(0, 1.0, 0.55); root.add(chamber);



// Burner Head & Nebulizer

const burnerHead=new T.Mesh(new T.BoxGeometry(0.8, 0.1, 0.15), mats.burner);

burnerHead.position.set(0, 0.7, 0.55); root.add(burnerHead);

const burnerBase=new T.Mesh(new T.CylinderGeometry(0.12, 0.12, 0.5, 16), mats.burner);

burnerBase.position.set(0, 0.4, 0.55); root.add(burnerBase);



// Flame Mesh

const flameGeo=new T.ConeGeometry(0.38, 0.7, 16);

flameGeo.rotateX(Math.PI);

const flameMesh=new T.Mesh(flameGeo, mats.flame);

flameMesh.position.set(0, 1.1, 0.55); root.add(flameMesh);



// Hollow Cathode Lamp (Cu HCL) Turret

const turret=new T.Mesh(new T.CylinderGeometry(0.5, 0.5, 0.3, 16), mats.dark);

turret.position.set(-1.4, 1.2, 0); root.add(turret);

const hcl=new T.Mesh(new T.CylinderGeometry(0.08, 0.08, 0.6, 16), mats.hcl);

hcl.rotation.z=Math.PI/2; hcl.position.set(-1.1, 1.2, 0); root.add(hcl);



// Monochromator & Detector Block

const detector=new T.Mesh(new T.BoxGeometry(0.6, 0.6, 0.6), mats.dark);

detector.position.set(1.5, 1.2, 0); root.add(detector);



// Light Beam (324.7 nm Line)

const beamGeo=new T.CylinderGeometry(0.02, 0.02, 2.6, 16);

beamGeo.rotateZ(Math.PI/2);

const beamMesh=new T.Mesh(beamGeo, mats.beam);

beamMesh.position.set(0.2, 1.2, 0.275); root.add(beamMesh);



// Atom Particles Group inside Flame

const atomGroup=new T.Group(); root.add(atomGroup);

const atomParticles=[];

for(let i=0; i<35; i++){

  const p=new T.Mesh(new T.SphereGeometry(0.02, 8, 8), mats.atom);

  p.position.set((Math.random()-.5)*0.6, 0.8+Math.random()*0.5, 0.55+(Math.random()-.5)*0.1);

  atomGroup.add(p);

  atomParticles.push(p);

}



/* --- Camera Controls --- */

let cs={th:.3, ph:1.2, r:6, tg:new T.Vector3(0,1,0)};

const V={

  front:[0, 1.2, 6, [0,1,0]],

  burner:[0.2, 1.3, 2.5, [0,0.9,0.55]],

  lamp:[-0.8, 1.3, 2.8, [-1.1,1.2,0]],

  reset:[.3, 1.2, 6, [0,1,0]]

};

function goCam(k){const v=V[k]||V.reset; cs.th=v[0]; cs.ph=v[1]; cs.r=v[2]; cs.tg.set(...v[3])}

document.querySelectorAll('[data-cam]').forEach(b=>b.onclick=()=>goCam(b.dataset.cam));



/* --- Interactivity --- */

$('#bFlame').onclick=()=>{S.flame=!S.flame; $('#bFlame').classList.toggle('on',S.flame)};

$('#bAspirate').onclick=()=>{S.aspirating=!S.aspirating; $('#bAspirate').classList.toggle('on',S.aspirating)};

$('#bBeam').onclick=()=>{S.beam=!S.beam; $('#bBeam').classList.toggle('on',S.beam)};



$('#m0').onclick=()=>{S.mode=0; setMode()};

$('#m1').onclick=()=>{S.mode=1; setMode()};

$('#m2').onclick=()=>{S.mode=2; setMode()};



function setMode(){

  $('#m0').classList.toggle('on', S.mode===0);

  $('#m1').classList.toggle('on', S.mode===1);

  $('#m2').classList.toggle('on', S.mode===2);

  if(S.mode===1) goCam('burner');

  else if(S.mode===2) goCam('lamp');

  else goCam('reset');

}



// Side Panel UI Rendering

const side=$('#side');

let currentTab='inst';

document.querySelectorAll('.tab').forEach(t=>{

  t.onclick=()=>{

    currentTab=t.dataset.t;

    document.querySelectorAll('.tab').forEach(b=>b.classList.remove('on'));

    t.classList.add('on');

    renderSide();

  }

});



function renderSide(){

  if(currentTab==='inst'){

    side.innerHTML=`

      <h3>Shimadzu AA-7000 Specifications</h3>

      <div class="box">

        <b>System:</b> Flame / Furnace Atomic Absorption Spectrophotometer<br>

      <h3>Instrument Controls</h3>

      <div class="box">

      </div>

    `;

    $('#cuConc').oninput=(e)=>{

      S.conc=parseFloat(e.target.value);

      $('#concVal').textContent=S.conc+' ppm';

    }

  } else if(currentTab==='atom'){

    side.innerHTML=`

      <h3>Copper (Cu) Atomization Process</h3>

      <div class="box">

        <b>1. Nebulization:</b> Liquid Cu²⁺ sample is drawn via capillary into fine droplets aerosol.<br><br>

        <small>*Absorbance is directly proportional to free ground state Cu atom density in the flame.</small>

      </div>

    `;

  }

}

renderSide();



/* --- Render Loop --- */

function loop(){

  requestAnimationFrame(loop);

  

  // Flame State & Color Shift with Copper aspirate

  flameMesh.visible = S.flame;

  if(S.flame){

    flameMesh.material = S.aspirating ? mats.flameCu : mats.flame;

    flameMesh.scale.set(1 + Math.sin(Date.now()*0.02)*0.05, 1 + Math.cos(Date.now()*0.015)*0.08, 1);

  }

  

  // Beam Visibility

  beamMesh.visible = S.beam;

  

  // Atom Particles Animation

  atomGroup.visible = S.flame && S.aspirating;

  if(atomGroup.visible){

    atomParticles.forEach((p, idx)=>{

      p.position.y += 0.015;

      if(p.position.y > 1.4){

        p.position.y = 0.8;

        p.position.x = (Math.random()-.5)*0.6;

      }

    });

  }



  // Camera Matrix Update

  cam.aspect = cv.parentElement.clientWidth / cv.parentElement.clientHeight;

  cam.updateProjectionMatrix();

  cam.position.set(

    cs.tg.x + cs.r * Math.sin(cs.ph) * Math.sin(cs.th),

    cs.tg.y + cs.r * Math.cos(cs.ph),

    cs.tg.z + cs.r * Math.sin(cs.ph) * Math.cos(cs.th)

  );

  cam.lookAt(cs.tg);

  

  R.setSize(cv.parentElement.clientWidth, cv.parentElement.clientHeight, false);

  R.render(scene, cam);

}

loop();
